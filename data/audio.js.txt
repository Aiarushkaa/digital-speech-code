
const TRACKS = [
  { bpm: 112, notes: [261.63, 329.63, 392, 523.25], bass: 65.41 },
  { bpm: 126, notes: [293.66, 369.99, 440, 587.33], bass: 73.42 },
  { bpm: 88, notes: [261.63, 329.63, 392, 493.88], bass: 65.41 },
  { bpm: 118, notes: [220, 277.18, 329.63, 440], bass: 55 },
  { bpm: 138, notes: [196, 246.94, 293.66, 392], bass: 49 }
];

export class GameAudio {
  constructor() {
    this.ctx = null;
    this.master = null;
    this.musicBus = null;
    this.sfxBus = null;

    this.musicEnabled = true;
    this.sfxEnabled = true;
    this.voiceEnabled = true;

    this.volume = 0.65;
    this.world = 0;
    this.stepTimer = null;
    this.musicTimer = null;
    this.nextBeat = 0;
    this.beat = 0;
    this.started = false;
  }

  async start() {
    if (!this.ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return false;

      this.ctx = new AC();

      this.master = this.ctx.createGain();
      this.musicBus = this.ctx.createGain();
      this.sfxBus = this.ctx.createGain();

      this.master.gain.value = this.volume;
      this.musicBus.gain.value = 0.55;
      this.sfxBus.gain.value = 0.9;

      this.musicBus.connect(this.master);
      this.sfxBus.connect(this.master);
      this.master.connect(this.ctx.destination);
    }

    await this.ctx.resume();
    this.started = true;

    this.launchEffect();

    if (this.musicEnabled) this.startMusic();
    return true;
  }

  setWorld(index) {
    this.world = Math.max(0, Math.min(4, Number(index) || 0));
    this.beat = 0;
    if (this.started && this.musicEnabled) this.startMusic();
  }

  setVolume(value) {
    this.volume = Math.max(0, Math.min(1, value));
    if (this.master) this.master.gain.setTargetAtTime(
      this.volume, this.ctx.currentTime, 0.1
    );
  }

  setMusicEnabled(value) {
    this.musicEnabled = Boolean(value);
    if (!this.started) return;
    if (this.musicEnabled) this.startMusic();
    else this.stopMusic();
  }

  setSfxEnabled(value) {
    this.sfxEnabled = Boolean(value);
  }

  setVoiceEnabled(value) {
    this.voiceEnabled = Boolean(value);
    if (!this.voiceEnabled) this.stopSpeech();
  }

  tone(freq, duration, type = 'sine', volume = 0.15, bus = this.sfxBus, when = 0) {
    if (!this.ctx) return;

    const t = this.ctx.currentTime + when;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = type;
    osc.frequency.setValueAtTime(freq, t);

    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.exponentialRampToValueAtTime(volume, t + 0.025);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + duration);

    osc.connect(gain);
    gain.connect(bus);

    osc.start(t);
    osc.stop(t + duration + 0.04);
  }

  noise(duration, volume, filterFrequency = 1200, when = 0) {
    if (!this.ctx) return;

    const length = Math.floor(this.ctx.sampleRate * duration);
    const buffer = this.ctx.createBuffer(1, length, this.ctx.sampleRate);
    const data = buffer.getChannelData(0);

    for (let i = 0; i < length; i++) {
      data[i] = Math.random() * 2 - 1;
    }

    const source = this.ctx.createBufferSource();
    const filter = this.ctx.createBiquadFilter();
    const gain = this.ctx.createGain();
    const t = this.ctx.currentTime + when;

    source.buffer = buffer;
    filter.type = 'lowpass';
    filter.frequency.value = filterFrequency;

    gain.gain.setValueAtTime(volume, t);
    gain.gain.exponentialRampToValueAtTime(0.001, t + duration);

    source.connect(filter);
    filter.connect(gain);
    gain.connect(this.sfxBus);

    source.start(t);
    source.stop(t + duration);
  }

  launchEffect() {
    if (!this.sfxEnabled) return;

    // Электронный запуск игры
    this.noise(1.2, 0.22, 1800);
    this.tone(55, 1.3, 'sawtooth', 0.16, this.sfxBus);
    this.tone(110, 0.9, 'sine', 0.12, this.sfxBus, 0.25);

    [220, 330, 440, 660, 880].forEach((f, i) => {
      this.tone(f, 0.5, 'sawtooth', 0.07, this.sfxBus, 0.35 + i * 0.12);
    });
  }

  startMusic() {
    this.stopMusic();
    if (!this.started || !this.musicEnabled) return;

    const track = TRACKS[this.world];
    const beatLength = 60 / track.bpm;

    this.nextBeat = this.ctx.currentTime + 0.08;
    this.beat = 0;

    const schedule = () => {
      while (this.nextBeat < this.ctx.currentTime + 0.15) {
        this.playBeat(track, this.beat, this.nextBeat);
        this.nextBeat += beatLength / 2;
        this.beat++;
      }
    };

    schedule();
    this.musicTimer = setInterval(schedule, 50);
  }

  playBeat(track, beat, time) {
    const step = beat % 8;

    // Бас
    if (step === 0 || step === 4) {
      this.tone(
        track.bass,
        0.32,
        'sawtooth',
        0.09,
        this.musicBus,
        Math.max(0, time - this.ctx.currentTime)
      );
    }

    // Ритмичный удар
    if (step === 0 || step === 4) {
      this.kick(time);
    }

    // Перкуссия
    if (step === 2 || step === 6) {
      this.noiseAt(time, 0.045, 4500);
    }

    // Мелодия
    if (step % 2 === 0) {
      const note = track.notes[(beat + this.world) % track.notes.length];
      this.tone(
        note,
        0.22,
        'triangle',
        0.045,
        this.musicBus,
        Math.max(0, time - this.ctx.currentTime)
      );
    }
  }

  kick(time) {
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    const t = Math.max(time, this.ctx.currentTime);

    osc.type = 'sine';
    osc.frequency.setValueAtTime(120, t);
    osc.frequency.exponentialRampToValueAtTime(35, t + 0.16);

    gain.gain.setValueAtTime(0.32, t);
    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.18);

    osc.connect(gain);
    gain.connect(this.musicBus);

    osc.start(t);
    osc.stop(t + 0.2);
  }

  noiseAt(time, volume, frequency) {
    const delay = Math.max(0, time - this.ctx.currentTime);
    this.noise(0.12, volume, frequency, delay);
  }

  stopMusic() {
    if (this.musicTimer) {
      clearInterval(this.musicTimer);
      this.musicTimer = null;
    }
  }

  // Шаг игрока: вызывается из игрового цикла движения
  step() {
    if (!this.started || !this.sfxEnabled) return;

    this.noise(0.11, 0.19, 500);
    this.tone(75, 0.09, 'sine', 0.12);
  }

  // Правильный ответ
  correct() {
    if (!this.started || !this.sfxEnabled) return;

    [523, 659, 784, 1046].forEach((f, i) => {
      this.tone(f, 0.35, 'sine', 0.2, this.sfxBus, i * 0.09);
    });
  }

  // Неправильный ответ
  wrong() {
    if (!this.started || !this.sfxEnabled) return;

    this.tone(220, 0.35, 'sawtooth', 0.17);
    this.tone(165, 0.4, 'sawtooth', 0.15, this.sfxBus, 0.18);
  }

  coin() {
    if (!this.started || !this.sfxEnabled) return;

    [880, 1174, 1568].forEach((f, i) => {
      this.tone(f, 0.2, 'sine', 0.16, this.sfxBus, i * 0.07);
    });
  }

  achievement() {
    if (!this.started || !this.sfxEnabled) return;

    [523, 659, 784, 1046, 1318].forEach((f, i) => {
      this.tone(f, 0.45, 'triangle', 0.18, this.sfxBus, i * 0.13);
    });
  }

  portal() {
    if (!this.started || !this.sfxEnabled) return;

    this.noise(0.9, 0.25, 2400);
    [110, 220, 440, 880].forEach((f, i) => {
      this.tone(f, 0.65, 'sine', 0.14, this.sfxBus, i * 0.14);
    });
  }

  speak(message) {
    if (!this.voiceEnabled || !('speechSynthesis' in window)) return;

    this.stopSpeech();

    const utterance = new SpeechSynthesisUtterance(message);
    utterance.lang = 'ru-RU';
    utterance.rate = 0.92;
    utterance.pitch = 1;
    utterance.volume = 1;

    const voice = speechSynthesis.getVoices().find(
      v => v.lang.toLowerCase().startsWith('ru')
    );

    if (voice) utterance.voice = voice;
    speechSynthesis.speak(utterance);
  }

  stopSpeech() {
    if ('speechSynthesis' in window) speechSynthesis.cancel();
  }

  destroy() {
    this.stopMusic();
    this.stopSpeech();

    if (this.ctx) {
      this.ctx.close();
      this.ctx = null;
    }

    this.started = false;
  }
}
