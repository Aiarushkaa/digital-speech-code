
export class GameAudio {
  constructor() {
    this.ctx = null;
    this.master = null;
    this.musicBus = null;
    this.sfxBus = null;

    this.started = false;
    this.musicEnabled = true;
    this.sfxEnabled = true;
    this.voiceEnabled = true;
    this.volume = 0.7;

    this.musicTimer = null;
    this.musicStep = 0;
    this.worldIndex = 0;

    this.tracks = [
      { name: 'city', notes: [261, 329, 392, 329, 293, 349, 440, 349], wave: 'sine', tempo: 380 },
      { name: 'memes', notes: [392, 440, 523, 440, 349, 392, 587, 523], wave: 'triangle', tempo: 260 },
      { name: 'campus', notes: [293, 349, 440, 392, 329, 392, 494, 440], wave: 'sine', tempo: 420 },
      { name: 'laboratory', notes: [220, 277, 330, 415, 330, 277, 370, 440], wave: 'triangle', tempo: 340 },
      { name: 'arena', notes: [196, 247, 294, 370, 294, 247, 330, 392], wave: 'sawtooth', tempo: 300 }
    ];
  }

  async start() {
    if (!this.ctx) {
      const AudioContextClass =
        window.AudioContext || window.webkitAudioContext;

      if (!AudioContextClass) {
        throw new Error('Web Audio API не поддерживается');
      }

      this.ctx = new AudioContextClass();

      this.master = this.ctx.createGain();
      this.musicBus = this.ctx.createGain();
      this.sfxBus = this.ctx.createGain();

      this.master.gain.value = this.volume;
      this.musicBus.gain.value = 0.22;
      this.sfxBus.gain.value = 0.8;

      this.musicBus.connect(this.master);
      this.sfxBus.connect(this.master);
      this.master.connect(this.ctx.destination);
    }

    await this.ctx.resume();
    this.started = true;

    this.launchEffect();
    this.startMusic();
  }

  setVolume(value) {
    this.volume = Math.max(0, Math.min(1, value));

    if (this.master && this.ctx) {
      this.master.gain.setTargetAtTime(
        this.volume,
        this.ctx.currentTime,
        0.05
      );
    }
  }

  setMusicEnabled(enabled) {
    this.musicEnabled = enabled;

    if (this.musicBus && this.ctx) {
      this.musicBus.gain.setTargetAtTime(
        enabled ? 0.22 : 0,
        this.ctx.currentTime,
        0.08
      );
    }

    if (enabled) this.startMusic();
    else this.stopMusic();
  }

  setSfxEnabled(enabled) {
    this.sfxEnabled = enabled;

    if (this.sfxBus && this.ctx) {
      this.sfxBus.gain.setTargetAtTime(
        enabled ? 0.8 : 0,
        this.ctx.currentTime,
        0.05
      );
    }
  }

  setVoiceEnabled(enabled) {
    this.voiceEnabled = enabled;

    if (!enabled) this.stopSpeech();
  }

  setWorld(index) {
    this.worldIndex = Math.max(
      0,
      Math.min(this.tracks.length - 1, index)
    );

    this.musicStep = 0;

    if (this.musicEnabled && this.started) {
      this.stopMusic();
      this.startMusic();
    }
  }

  tone(
    frequency,
    duration = 0.2,
    type = 'sine',
    volume = 0.15,
    delay = 0,
    destination = this.sfxBus
  ) {
    if (!this.started || !this.ctx || !destination) return;
    if (destination === this.sfxBus && !this.sfxEnabled) return;

    const now = this.ctx.currentTime + delay;

    const oscillator = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    oscillator.type = type;
    oscillator.frequency.setValueAtTime(frequency, now);

    gain.gain.setValueAtTime(0.001, now);
    gain.gain.exponentialRampToValueAtTime(
      Math.max(0.001, volume),
      now + 0.02
    );
    gain.gain.exponentialRampToValueAtTime(
      0.001,
      now + duration
    );

    oscillator.connect(gain);
    gain.connect(destination);

    oscillator.start(now);
    oscillator.stop(now + duration + 0.03);
  }

  noise(
    duration = 0.2,
    volume = 0.12,
    filterFrequency = 1200,
    delay = 0,
    destination = this.sfxBus
  ) {
    if (!this.started || !this.ctx || !destination) return;
    if (destination === this.sfxBus && !this.sfxEnabled) return;

    const length = Math.floor(this.ctx.sampleRate * duration);
    const buffer = this.ctx.createBuffer(
      1,
      length,
      this.ctx.sampleRate
    );

    const data = buffer.getChannelData(0);

    for (let i = 0; i < length; i++) {
      data[i] = Math.random() * 2 - 1;
    }

    const source = this.ctx.createBufferSource();
    const filter = this.ctx.createBiquadFilter();
    const gain = this.ctx.createGain();

    source.buffer = buffer;
    filter.type = 'lowpass';
    filter.frequency.value = filterFrequency;

    const now = this.ctx.currentTime + delay;

    gain.gain.setValueAtTime(volume, now);
    gain.gain.exponentialRampToValueAtTime(
      0.001,
      now + duration
    );

    source.connect(filter);
    filter.connect(gain);
    gain.connect(destination);

    source.start(now);
    source.stop(now + duration);
  }

  startMusic() {
    if (!this.started || !this.musicEnabled || this.musicTimer) return;

    const playStep = () => {
      if (!this.started || !this.musicEnabled) {
        this.musicTimer = null;
        return;
      }

      const track = this.tracks[this.worldIndex];
      const note = track.notes[this.musicStep % track.notes.length];

      this.tone(
        note,
        0.32,
        track.wave,
        0.075,
        0,
        this.musicBus
      );

      if (this.musicStep % 2 === 0) {
        this.tone(
          note / 2,
          0.48,
          'sine',
          0.045,
          0,
          this.musicBus
        );
      }

      if (this.musicStep % 4 === 0) {
        this.noise(
          0.08,
          0.025,
          900,
          0,
          this.musicBus
        );
      }

      this.musicStep++;

      this.musicTimer = setTimeout(
        playStep,
        track.tempo
      );
    };

    playStep();
  }

  stopMusic() {
    if (this.musicTimer) {
      clearTimeout(this.musicTimer);
      this.musicTimer = null;
    }
  }

  launchEffect() {
    this.tone(180, 0.5, 'sawtooth', 0.12);
    this.tone(360, 0.6, 'sine', 0.1, 0.12);
    this.tone(720, 0.7, 'sine', 0.08, 0.25);
  }

  step() {
    this.noise(0.07, 0.08, 350);
    this.tone(75, 0.08, 'sine', 0.07);
  }

  correct() {
    this.tone(523, 0.18, 'sine', 0.16);
    this.tone(659, 0.2, 'sine', 0.16, 0.12);
    this.tone(784, 0.35, 'sine', 0.18, 0.24);
  }

  wrong() {
    this.tone(220, 0.25, 'sawtooth', 0.13);
    this.tone(165, 0.4, 'triangle', 0.12, 0.18);
  }

  coin() {
    this.tone(880, 0.12, 'sine', 0.13);
    this.tone(1175, 0.2, 'sine', 0.14, 0.1);
  }

  achievement() {
    this.tone(523, 0.2, 'triangle', 0.15);
    this.tone(659, 0.2, 'triangle', 0.15, 0.15);
    this.tone(784, 0.2, 'triangle', 0.17, 0.3);
    this.tone(1046, 0.6, 'sine', 0.18, 0.45);
  }

  portal() {
    this.tone(220, 0.5, 'sine', 0.13);
    this.tone(440, 0.5, 'triangle', 0.12, 0.12);
    this.tone(880, 0.7, 'sine', 0.1, 0.25);
    this.noise(0.5, 0.08, 1800);
  }

  speak(message) {
    if (!this.voiceEnabled || !('speechSynthesis' in window)) return;

    window.speechSynthesis.cancel();

    const utterance = new SpeechSynthesisUtterance(message);
    utterance.lang = 'ru-RU';
    utterance.rate = 0.92;
    utterance.pitch = 1;
    utterance.volume = 1;

    window.speechSynthesis.speak(utterance);
  }

  stopSpeech() {
    if ('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
    }
  }

  destroy() {
    this.stopMusic();
    this.stopSpeech();

    if (this.ctx && this.ctx.state !== 'closed') {
      this.ctx.close();
    }

    this.started = false;
    this.ctx = null;
  }
}
